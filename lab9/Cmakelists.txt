cmake_minimum_required(VERSION 3.16)

project(lab5 VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)

find_package(Qt6 COMPONENTS Core Gui Widgets Sql QUIET)
if (NOT Qt6_FOUND)
    find_package(Qt5 COMPONENTS Core Gui Widgets Sql REQUIRED)
endif()

set(SOURCES
    main.cpp
    shapes.cpp
    shapecanvas.cpp
    shapesdatabase.cpp
    mainwindow.cpp
    shapedelegate.cpp
)

set(HEADERS
    shapes.h
    shapecanvas.h
    shapesdatabase.h
    mainwindow.h
    shapedelegate.h
)

add_executable(main
    ${SOURCES}
    ${HEADERS}
)

if (Qt6_FOUND)
    target_link_libraries(main PRIVATE
        Qt6::Core
        Qt6::Gui
        Qt6::Widgets
        Qt6::Sql
    )
else()
    target_link_libraries(main PRIVATE
        Qt5::Core
        Qt5::Gui
        Qt5::Widgets
        Qt5::Sql
    )
endif()

install(TARGETS main
    BUNDLE DESTINATION .
    RUNTIME DESTINATION bin
)
